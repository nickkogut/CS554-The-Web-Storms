    friends: [{
        id:                 firebase uid,
        username:           string,
        friend_timestamp    (time of becoming friends): Date,
        lastInteracted      (most recent time of friend, send game request, ...): Date
    },
    ...

    ],
    quiz_history: [{
        name:               string
        moderator_id:       firebase uid,

        questions_total:    int,

        num_participants:   int,
        timestamp:          Date
    },
    ...
    ]
}
